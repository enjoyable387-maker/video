  function spaceBr(el) {
    $$('br', el).forEach(function (br) {
      var p = br.previousSibling;
      if (p && p.nodeType === 3 && /\s$/.test(p.nodeValue)) return;
      br.parentNode.insertBefore(document.createTextNode(' '), br);
    });
  }

  /* 단어 감싸기 — 줄 단위 리빌과 흐림 리빌이 같이 쓴다 */
  function splitWords(el) {
    var n = 0;
    var walk = function (node) {
      Array.prototype.slice.call(node.childNodes).forEach(function (c) {
        if (c.nodeType === 3) {
          var frag = document.createDocumentFragment();
          c.nodeValue.split(/(\s+)/).forEach(function (part) {
            if (!part) return;
            if (/^\s+$/.test(part)) { frag.appendChild(document.createTextNode(part)); return; }
            var w = document.createElement('span'); w.className = 'dt-w';
            var i = document.createElement('span'); i.textContent = part; w.style.setProperty('--i', n++);
            w.appendChild(i); frag.appendChild(w);
          });
          c.parentNode.replaceChild(frag, c);
        } else if (c.nodeType === 1 && c.tagName !== 'BR') walk(c);
      });
    };
    walk(el);
  }

  function init(root) {
    var key = root.getAttribute('data-dt-key') || 'main';
    html.classList.add('dt-on');
    hideHost();
    freeOverflow(root);
    chrome(key);
    $$('[data-dt-src]', root).forEach(function (img) { img.src = media(img.getAttribute('data-dt-src')); });
    $$('[data-dt-page]', root).forEach(function (a) { a.href = page(a.getAttribute('data-dt-page')) + (a.getAttribute('data-dt-hash') || ''); });
    hero(root);
    site(root);
    houses(root);
    chart(root);
    gantt(root);
    worksPage(root);
    gallery(root);
    ask(root);
    spaceBr(root);

    /* ── 등장 ── */
    var lrEls = $$('.dt-lr, .dt-blur', root);
    lrEls.forEach(splitWords);
    var rvs = $$('.dt-rv, .dt-chart, .dt-gantt, .dt-cost', root);
    var counts = $$('[data-dt-count]', root);
    if (reduce || !('IntersectionObserver' in window)) {
      lrEls.forEach(function (el) { el.classList.add('dt-go'); });
      rvs.forEach(function (el) { el.classList.add('dt-in'); });
      return;
    }
    var lio = new IntersectionObserver(function (es) {
      es.forEach(function (en) { if (en.isIntersecting) { lio.unobserve(en.target); lines(en.target); requestAnimationFrame(function () { en.target.classList.add('dt-go'); }); } });
    }, { threshold: .3, rootMargin: '0px 0px -8% 0px' });
    lrEls.forEach(function (el) { lio.observe(el); });
    var io = new IntersectionObserver(function (es) {
      es.forEach(function (en, k) { if (en.isIntersecting) { io.unobserve(en.target); setTimeout(function () { en.target.classList.add('dt-in'); }, k * 90); } });
    }, { threshold: .12 });
    rvs.forEach(function (el) { io.observe(el); });
    var cio = new IntersectionObserver(function (es) {
      es.forEach(function (en) {
        if (!en.isIntersecting) return;
        cio.unobserve(en.target);
        var el = en.target, to = +el.getAttribute('data-dt-count'), t0 = 0;
        var fmt = function (v) { return String(v).replace(/\B(?=(\d{3})+(?!\d))/g, ','); };
        var step = function (t) { if (!t0) t0 = t; var k = clamp((t - t0) / 1400, 0, 1); el.textContent = fmt(Math.round(to * ease(k))); if (k < 1) requestAnimationFrame(step); };
        requestAnimationFrame(step);
      });
    }, { threshold: .5 });
    counts.forEach(function (el) { cio.observe(el); });
  }

  /* 위젯이 늦게 그려져도 나타나는 순간 시작 (최대 8초 감시) */
  function start() { var root = $('.dt-page'); if (!root || root.getAttribute('data-dt-init')) return !!root; root.setAttribute('data-dt-init', '1'); init(root); return true; }
  if (!start()) {
    var mo = new MutationObserver(function () { if (start()) mo.disconnect(); });
    mo.observe(html, { childList: true, subtree: true });
    setTimeout(function () { mo.disconnect(); }, 8000);
  }
